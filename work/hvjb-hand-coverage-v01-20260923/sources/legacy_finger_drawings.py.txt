def text(ax, x, y, value, size=13, color=INK, bold=False, ha="left", va="center", **kwargs):
    return ax.text(x, y, value, fontsize=size, fontproperties=BOLD if bold else FONT,
                   color=color, ha=ha, va=va, linespacing=1.55, **kwargs)


def rect(ax, x, y, w, h, color=BLUE, edge=None, **kwargs):
    ax.add_patch(Rectangle((x, y), w, h, facecolor=color, edgecolor=edge or color, lw=1.4, **kwargs))


def circle(ax, x, y, r, color=METAL, edge=GRAY, **kwargs):
    ax.add_patch(Circle((x, y), r, facecolor=color, edgecolor=edge, lw=1.5, **kwargs))


def arrow(ax, start, end, color=BLUE, both=False, **kwargs):
    ax.add_patch(FancyArrowPatch(start, end, arrowstyle="<->" if both else "-|>",
                               mutation_scale=15, color=color, lw=1.8, **kwargs))


def callout(ax, anchor, endpoint, label, color=INK, align="left"):
    ax.plot([anchor[0], endpoint[0]], [anchor[1], endpoint[1]], color=color, lw=1)
    circle(ax, anchor[0], anchor[1], .65, color, color)
    text(ax, endpoint[0], endpoint[1] + 2, label, 12, color, ha=align, va="bottom")

def panel(fig, rect_xywh, title, limits=(100, 70)):
    x, y, w, h = rect_xywh
    ax = fig.add_axes([x/W, (H-y-h)/H, w/W, h/H])
    ax.set(xlim=(0, limits[0]), ylim=(0, limits[1])); ax.axis("off")
    ax.set_aspect("equal")
    text(ax, 1, limits[1]-2, title, 13, bold=True, va="top")
    return ax


def picture(fig, name, area, title):
    x, y, w, h = area
    ax = fig.add_axes([x/W, (H-y-h)/H, w/W, h/H])
    ax.imshow(Image.open(ROOT / "figures" / (name + ".png")))
    ax.axis("off")
    ax.set_title(title, fontproperties=BOLD, color=INK, fontsize=12, loc="left", pad=9)


def notes(ax, sections):
    rect(ax, 1110, 190, 435, 752, PALE)
    y = 215
    for title, body in sections:
        text(ax, 1131, y, title, 13, BLUE, True, va="top")
        y += 36
        lines = []
        for paragraph in body.split("\n"):
            current, width = "", 0
            for token in re.findall(r"[A-Za-z0-9][A-Za-z0-9./×°+\-]*|.", paragraph):
                token_width = sum(1 if unicodedata.east_asian_width(ch) in "WF" else .55 for ch in token)
                if current and width + token_width > 24 and token not in "。、，．）」":
                    lines.append(current)
                    current, width = "", 0
                current += token
                width += token_width
            lines.append(current)
        text(ax, 1131, y, "\n".join(lines), 12, va="top")
        y += len(lines) * 25 + 27
    assert y <= 966, y


def source(ax, content):
    text(ax, 58, 967, "保存仕様：" + content, 9, GRAY)


def save(pdf, fig, name):
    pdf.savefig(fig)
    output = PNG / f"{len(MANIFEST)+1:02d}_{name}.png"
    fig.savefig(output, dpi=160)
    MANIFEST.append({"page": len(MANIFEST)+1, "png": str(output.relative_to(ROOT)), "label": name})
    plt.close(fig)
